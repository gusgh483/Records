    {
        print("OnEnable"); // start() 직전에 호출, 객체를 다시 활성화해도 호출
    }

    private void OnDisable()
    {
        // 객체 비활성화, 게임 오브젝트가 제거될때도 호출
        // Destroy로 객체 삭제나 게임 종료 시 제거됨
        print("OnDisable"); 
    }

    private void Update()
    {
        print("Update");
    }
